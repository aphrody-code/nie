# ── WASM-specific build configuration ──────────────────────────────
# Minimal dependencies for WebAssembly compilation
# Emscripten toolchain required: $EMSDK/upstream/emscripten/cmake/Modules/Platform/Emscripten.cmake

if(NOT EMSCRIPTEN)
  message(FATAL_ERROR "WASM build requires Emscripten toolchain. Set CMAKE_TOOLCHAIN_FILE to emscripten.cmake")
endif()

# ── WASM optimization flags ────────────────────────────────────────
add_compile_options(
  -sWASM=1
  -sUSE_PTHREADS=0
  -fwasm-exceptions
  -fPIC
)

add_link_options(
  -sWASM=1
  -sALLOW_MEMORY_GROWTH=1
  -sINITIAL_MEMORY=64mb
  -sEXPORTED_RUNTIME_METHODS=['UTF8ToString','stringToUTF8','ccall','cwrap']
  -sEXPORT_ALL=0
)

# ── WASM-only feature flags ────────────────────────────────────────
set(IECODE_WASM ON CACHE BOOL "Building for WebAssembly")
set(IECODE_NO_GUI ON)              # No D3D11 in WASM
set(IECODE_NO_PHYSICS ON)          # No Bullet3
set(IECODE_NO_3D_RENDER ON)        # No bgfx/DirectXTex
set(IECODE_NO_LUA ON)              # Optional: Lua/Sol2
set(IECODE_UNITY_BUILD ON)         # Faster WASM compilation
set(IECODE_LTO OFF)                # Emscripten handles optimization

# ── WASM-compatible core libraries ─────────────────────────────────
find_package(fmt            CONFIG REQUIRED)
find_package(nlohmann_json  CONFIG REQUIRED)
find_package(spdlog         CONFIG REQUIRED)

# ── Disable platform-specific code ─────────────────────────────────
add_compile_definitions(
  IECODE_WASM=1
  NDEBUG
  _GLIBCXX_ASSERTIONS=0
)

# ── Build only core extraction/conversion libraries ────────────────
# iecode_core: compression, crypto, VFS, parsers
# iecode_level5: G4TX, G4MG, cfg.bin parsing
# iecode_criware: CPK, UTF handling (file-independent)
